add_internal_library(RHI
    TYPE                STATIC
    SOURCES             ${SOURCES_LIST}
    INCLUDES            "${CMAKE_CURRENT_SOURCE_DIR}/includes"
    PRIVATE_INCLUDES    "${CMAKE_CURRENT_SOURCE_DIR}/sources"
    PUBLIC_DEPENDENCY   Core::Core Module::GenericApplication
    ALIAS               Module::RHI
)

target_compile_definitions(RHI PRIVATE
    RHI_ENABLE_VALIDATION=$<BOOL:${RHI_ENABLE_VALIDATION}>
    RHI_ENABLE_PIPELINE_STATISTICS=$<BOOL:${RHI_ENABLE_PIPELINE_STATISTICS}>
)

if(GRAPHICS_API_DX12)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_DX12=1)
    target_include_directories(RHI PRIVATE ${DX12_INCLUDE_DIR})
    target_link_libraries(RHI PRIVATE ${DX12_D3D_LIBRARY} ${DXGI_LIBRARY})

elseif(GRAPHICS_API_DX11)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_DX11=1)
    target_include_directories(RHI PRIVATE ${DX11_INCLUDE_DIR})
    target_link_libraries(RHI PRIVATE ${DX11_D3D_LIBRARY} ${DXGI_LIBRARY})

elseif(GRAPHICS_API_DX10)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_DX10=1)
    target_include_directories(RHI PRIVATE ${DX10_INCLUDE_DIR})
    target_link_libraries(RHI PRIVATE ${DX10_D3D_LIBRARY} ${DXGI_LIBRARY})

elseif(GRAPHICS_API_DX9)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_DX9=1)
    target_include_directories(RHI PRIVATE ${DX9_INCLUDE_DIR})
    target_link_libraries(RHI PRIVATE ${DX9_D3D_LIBRARY})

elseif(GRAPHICS_API_VULKAN)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_VULKAN=1)
    if(USE_SDL)
        target_compile_definitions(RHI PRIVATE USE_SDL=1)
        target_link_libraries(RHI PRIVATE SDL3::SDL3)
    else()
        message(FATAL_ERROR "Vulkan RHI currently requires the SDL generic window backend")
    endif()
    # 专门伺候 <windows.h> 这个头文件:
    # VK_USE_PLATFORM_WIN32_KHR: 用于启动 win32 surface
    # NOMINMAX: 避免 windows.h 定义 min/max 宏
    # WIN32_LEAN_AND_MEAN: 避免 windows.h 引入过多不必要的头文件
    if (WIN32)
        target_compile_definitions(RHI PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(TARGET Vulkan::Vulkan)
        target_link_libraries(RHI PRIVATE Vulkan::Vulkan)
    else()
        target_include_directories(RHI PRIVATE ${Vulkan_INCLUDE_DIRS})
        target_link_libraries(RHI PRIVATE ${Vulkan_LIBRARIES})
    endif()

elseif(GRAPHICS_API_METAL)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_METAL=1)
    target_link_libraries(RHI PRIVATE ${METAL_LIBRARY})

elseif(GRAPHICS_API_OPENGL)
    target_compile_definitions(RHI PRIVATE GRAPHICS_API_OPENGL=1)
    if(TARGET OpenGL::GL)
        target_link_libraries(RHI PRIVATE OpenGL::GL)
    else()
        target_include_directories(RHI PRIVATE ${OPENGL_INCLUDE_DIRS})
        target_link_libraries(RHI PRIVATE ${OPENGL_LIBRARIES})
    endif()

else()
    message(FATAL_ERROR "未找到任何受支持的图形后端")
endif()
